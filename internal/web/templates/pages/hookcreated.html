systemd: fetch daily and reload

1. Token file, readable by root only:

install -d -m 700 /etc/pki-portal
printf 'PKI_TOKEN=%s\n' '{{.Token}}' > /etc/pki-portal/{{.Slug}}.env
chmod 600 /etc/pki-portal/{{.Slug}}.env

2. /etc/systemd/system/pki-fetch-{{.Slug}}.service

[Unit]
Description=Fetch {{index .Cert.Names 0}} from the PKI portal
Wants=network-online.target
After=network-online.target

[Service]
Type=oneshot
EnvironmentFile=/etc/pki-portal/{{.Slug}}.env
ExecStart=/bin/sh -c 'set -e; u={{.BaseURL}}/api/v1/certs/{{.Cert.ID}}; \
  curl -fsS -H "Authorization: Bearer $PKI_TOKEN" -o /etc/ssl/{{.Slug}}.pem.new $u/fullchain; \{{if .HasKey}}
  curl -fsS -H "Authorization: Bearer $PKI_TOKEN" -o /etc/ssl/{{.Slug}}.key.new $u/key; \
  chmod 600 /etc/ssl/{{.Slug}}.key.new; mv /etc/ssl/{{.Slug}}.key.new /etc/ssl/{{.Slug}}.key; \{{end}}
  mv /etc/ssl/{{.Slug}}.pem.new /etc/ssl/{{.Slug}}.pem'
# Change to whatever uses the certificate:
ExecStartPost=/bin/systemctl reload nginx

3. /etc/systemd/system/pki-fetch-{{.Slug}}.timer

[Unit]
Description=Daily fetch of {{index .Cert.Names 0}}

[Timer]
OnCalendar=daily
RandomizedDelaySec=1h
Persistent=true

[Install]
WantedBy=timers.target
systemctl daemon-reload
systemctl enable --now pki-fetch-{{.Slug}}.timer
systemctl start pki-fetch-{{.Slug}}.service   # first fetch now
{{end}}
